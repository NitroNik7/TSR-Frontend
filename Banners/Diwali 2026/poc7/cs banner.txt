<script>
    document.addEventListener('DOMContentLoaded', function () {
        const tsrBannerCloseBtns = document.querySelectorAll('.tsrBannerCloseAction');
        const tsrBannerContainer = document.getElementById('tsrBannerMainContainer');

        if (tsrBannerCloseBtns.length > 0 && tsrBannerContainer) {
            tsrBannerCloseBtns.forEach(function (btn) {
                btn.addEventListener('click', function () {
                    tsrBannerContainer.style.transition = 'opacity 0.8s ease, transform 0.8s ease';
                    tsrBannerContainer.style.opacity = '0';
                    tsrBannerContainer.style.transform = 'translateX(-50%) translateY(200px)';

                    setTimeout(function () {
                        tsrBannerContainer.remove();
                    }, 800);
                });
            });
        }

        let delay = 30000; // in ms
        setTimeout(() => {
            let isPr = mtgv.mtpp.pr;
            // let isPr = false;
            if (!isPr) {
                tsrBannerContainer.style.display = "block";
                tsrBannerContainer.style.animation = "tsrBannerSlideUp 0.6s ease-out";
            }
        }, delay);
    });
</script>
